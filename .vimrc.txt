" 1. Explicitly enable Sneak's Label/Streak mode
let g:sneak#label = 1

" 2. Configure colors for matches and labels
function! FixSneakLabelsAndColors() abort
  " The targets / current cursor landing spot (Bright Green on Charcoal)
  highlight Sneak             guifg=#00FF66 guibg=#1E1E1E gui=bold ctermfg=green ctermbg=black
  highlight SneakCurrent      guifg=#00FF66 guibg=#1E1E1E gui=bold ctermfg=green ctermbg=black
  highlight SneakPluginTarget guifg=#00FF66 guibg=#1E1E1E gui=bold ctermfg=green ctermbg=black

  " The Label key letters to type (Bright Yellow on Charcoal so they pop out clearly)
  highlight SneakLabel        guifg=#FFFF00 guibg=#1E1E1E gui=bold,underline ctermfg=yellow ctermbg=black
  highlight SneakStreakTarget guifg=#FFFF00 guibg=#1E1E1E gui=bold ctermfg=yellow ctermbg=black

  " Hide original characters under the label without magenta bleeding
  highlight SneakLabelMask    guifg=#1E1E1E guibg=#1E1E1E ctermfg=black ctermbg=black
  highlight SneakStreakMask   guifg=#1E1E1E guibg=#1E1E1E ctermfg=black ctermbg=black
endfunction

" 3. Ensure colors apply on startup and theme reloads
augroup SneakFixFinal
  autocmd!
  autocmd ColorScheme,VimEnter * call FixSneakLabelsAndColors()
augroup END

call FixSneakLabelsAndColors()
